1 Setup

This report reads the three Parquet datasets, prints descriptive tables directly into the rendered report, and saves figures as PDF files in ../reports/figures.

Code 
library(tidyverse)
library(arrow)
library(knitr)

# Paths are relative to the directory containing this QMD file.
rounds_file <- "../data/analysis/rounds.parquet"
states_file <- "../data/analysis/round_states.parquet"
enriched_file <- "../data/analysis/round_states_enriched.parquet"

figures_dir <- "../reports/figures"
dir.create(figures_dir, showWarnings = FALSE, recursive = TRUE)

rounds <- read_parquet(rounds_file) |> as_tibble()
round_states <- read_parquet(states_file) |> as_tibble()
round_states_enriched <- read_parquet(enriched_file) |> as_tibble()

stopifnot(all(c("match_id", "map", "round_num") %in% names(rounds)))
stopifnot(all(c("match_id", "map", "round_num", "tick", "seconds",
                "ct_alive", "t_alive", "ct_win") %in% names(round_states)))
stopifnot(all(c("match_id", "map", "round_num") %in%
                names(round_states_enriched)))

# Save each ggplot as a PDF.
save_pdf <- function(plot, filename, width = 9, height = 6) {
  ggsave(
    filename = file.path(figures_dir, filename),
    plot = plot,
    width = width,
    height = height,
    device = grDevices::cairo_pdf
  )
}

# Consistent table display in the rendered report.
show_table <- function(x, digits = 2) {
  knitr::kable(x, digits = digits, format = "html")
}

4.3 Alive-player advantage and eventual outcome

This is a snapshot-level exploratory comparison. Because the same round contributes multiple snapshots, the percentages below should not be interpreted as independent-round estimates.

Code 
state_advantage <- round_states |>
  mutate(
    alive_difference = ct_alive - t_alive,
    alive_advantage = case_when(
      alive_difference > 0 ~ "CT advantage",
      alive_difference < 0 ~ "T advantage",
      alive_difference == 0 ~ "Equal alive players",
      TRUE ~ NA_character_
    )
  )

advantage_by_outcome <- state_advantage |>
  filter(!is.na(ct_win), !is.na(alive_advantage)) |>
  group_by(alive_advantage) |>
  summarise(
    snapshots = n(),
    ct_round_win_pct = 100 * mean(ct_win, na.rm = TRUE),
    .groups = "drop"
  )
show_table(advantage_by_outcome)
alive_advantage snapshots ct_round_win_pct
CT advantage 28565 83.42
Equal alive players 60682 50.68
T advantage 26954 10.40
Code 
p_alive_advantage <- ggplot(
  advantage_by_outcome, aes(x = alive_advantage, y = ct_round_win_pct)
) +
  geom_col() +
  geom_text(aes(label = sprintf("%.1f%%", ct_round_win_pct)), vjust = -0.3) +
  scale_y_continuous(limits = c(0, 100),
                     expand = expansion(mult = c(0, 0.08))) +
  labs(title = "CT round-win percentage by alive-player advantage",
       subtitle = "Snapshot-level descriptive comparison",
       x = "Player-count state", y = "CT round wins (%)") +

5.3 Check round-level fields for consistency

These fields should generally be constant across all snapshots belonging to the same round.

Code 
round_level_fields <- intersect(
  c("winner_side", "winner_team", "reason_name", "is_overtime",
    "is_pistol_round", "bomb_planted", "ct_buy_type", "t_buy_type",
    "first_death_side", "round_duration_s"),
  names(round_states_enriched)
)

round_constant_checks <- map_dfr(round_level_fields, function(field) {
  round_states_enriched |>
    group_by(match_id, round_num) |>
    summarise(
      distinct_values = n_distinct(.data[[field]], na.rm = FALSE),
      .groups = "drop"
    ) |>
    summarise(
      field = field,
      rounds_with_multiple_values = sum(distinct_values > 1),
      .groups = "drop"
    )
})

7.2 4.2 Alive-player advantage at 30 seconds

state_at_30s <- round_states_enriched |>
  filter(seconds <= 30) |>
  group_by(match_id, round_num) |>
  slice_max(order_by = seconds, n = 1, with_ties = FALSE) |>
  ungroup() |>
  mutate(
    alive_difference = ct_alive - t_alive,
    alive_advantage = case_when(
      alive_difference > 0 ~ "CT has more alive",
      alive_difference < 0 ~ "T has more alive",
      alive_difference == 0 ~ "Equal alive players",

    )
  )

8 Notes and interpretation

  • rounds.parquet contains round-level records; round_states.parquet contains multiple time snapshots per round.
  • round_states_enriched.parquet adds round-level information to each snapshot. The row count should normally remain the same as the original snapshot data.
  • Missing seconds_since_plant and plant_site values may be expected in rounds or snapshots where the bomb has not been planted.
  • The snapshot-level alive-advantage summary in section 2.2 weights rounds with more snapshots more heavily. Use the one-snapshot-per-round analysis in section 4.2 for a more comparable round-level description.
  • The 30-second analyses use the latest available snapshot at or before 30 seconds; it may not be exactly 30 seconds.
  • Figures are saved as PDF files in ../reports/figures.
